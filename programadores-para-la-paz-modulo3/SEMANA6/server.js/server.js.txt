server.js
const express = require('express');
const app = express();

app.use(express.json());

app.post('/registro', (req, res) => {

  const nombre = req.body.nombre;
  const mensaje = req.body.mensaje;

  res.json({
    estado: "Datos recibidos",
    nombre: nombre,
    mensaje: mensaje
  });

});

app.post('/incidencia', (req, res) => {

  const tipo = req.body.tipo;
  const descripcion = req.body.descripcion;

  res.json({
    mensaje: "Incidencia registrada",
    tipo: tipo,
    descripcion: descripcion
  });

});

app.listen(3000, () => {
  console.log('Servidor ejecutándose en puerto 3000');
});
prueba-api.txt
Al enviar una petición POST a la ruta http://localhost:3000/registro con el siguiente JSON:

{
  "nombre": "Maria",
  "mensaje": "Hola comunidad"
}

El servidor recibe los datos mediante req.body y responde con un objeto JSON que confirma la recepción de la información:

{
  "estado": "Datos recibidos",
  "nombre": "Maria",
  "mensaje": "Hola comunidad"
}

Esto demuestra que Express puede recibir y procesar datos enviados por el usuario en formato JSON.
